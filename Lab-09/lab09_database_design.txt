\echo '5. Library Management System: ER entities and relationships'
5. Library Management System: ER entities and relationships
-- Author WRITES Book: many-to-many via lab09_book_authors
-- Member BORROWS Book: one-to-many via lab09_loans
CREATE TABLE lab09_authors (
 author_id SERIAL PRIMARY KEY,
 first_name VARCHAR(50) NOT NULL,
 last_name VARCHAR(50) NOT NULL,
 birth_date DATE
);
CREATE TABLE
CREATE TABLE lab09_books (
 book_id SERIAL PRIMARY KEY,
 title VARCHAR(200) NOT NULL,
 isbn VARCHAR(13) UNIQUE,
 publication_year INTEGER,
 available_copies INTEGER NOT NULL DEFAULT 1 CHECK(available_copies>=0)
);
CREATE TABLE
CREATE TABLE lab09_book_authors (
 book_id INTEGER REFERENCES lab09_books(book_id) ON DELETE CASCADE,
 author_id INTEGER REFERENCES lab09_authors(author_id) ON DELETE CASCADE,
 PRIMARY KEY(book_id,author_id)
);
CREATE TABLE
CREATE TABLE lab09_members (
 member_id SERIAL PRIMARY KEY,
 first_name VARCHAR(50) NOT NULL,
 last_name VARCHAR(50) NOT NULL,
 email VARCHAR(100) UNIQUE NOT NULL,
 phone VARCHAR(15),
 membership_date DATE DEFAULT CURRENT_DATE
);
CREATE TABLE
CREATE TABLE lab09_loans (
 loan_id SERIAL PRIMARY KEY,
 member_id INTEGER NOT NULL REFERENCES lab09_members(member_id),
 book_id INTEGER NOT NULL REFERENCES lab09_books(book_id),
 loan_date DATE DEFAULT CURRENT_DATE,
 due_date DATE NOT NULL,
 return_date DATE,
 late_fee DECIMAL(10,2) DEFAULT 0.00 CHECK(late_fee>=0)
);
CREATE TABLE
